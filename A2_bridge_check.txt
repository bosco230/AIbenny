  59:     """Process-wide WebSocket fan-out.
  60: 
  61:     Implements the structural ``WSBridge`` Protocol (``async def
  62:     publish(message: dict) -> None``) without importing it from the
  63:     avatar package â€” keeps this file free of sibling-package imports.
  64: 
  65:     Each connected socket is registered via :meth:`add_socket` /
  66:     :meth:`remove_socket` and receives every ``publish`` call. A failed
  67:     send is logged and removes the offending socket from the fan-out
  68:     set so a dead client doesn't poison the broadcast.
  69:     """

  72:         self._sockets: set[WebSocket] = set()
  73:         self._send_locks: dict[WebSocket, asyncio.Lock] = {}

  76:         self._latest_bubble: dict[str, Any] | None = None

  84:         async with self._lock:
  85:             self._sockets.add(ws)
  86:             self._send_locks.setdefault(ws, asyncio.Lock())
  87:             replay = [message for message in (self._latest_runtime,) if message is not None]
  88:             replay.extend(self._component_health.values())
  89:             replay.extend(
  90:                 message
  91:                 for message in (self._latest_turn, self._latest_avatar, self._latest_bubble)

  93:             )

  97:             if not await self.send(ws, message):
  98:                 return False
  99:         return True

 102:         async with self._lock:
