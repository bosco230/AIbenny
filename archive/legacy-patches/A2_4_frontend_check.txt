   1: /**
   2:  * `WSProvider` — React context wrapping a `/ws` connection.
   3:  *
   4:  * Responsibilities:
   5:  *
   6:  * - Open a connection on mount via the injected `socketFactory`.
   7:  * - Auto-reconnect with jittered exponential backoff via
   8:  *   `createReconnectController`.
   9:  * - Decode every inbound message string into a typed `ServerMessage`
  10:  *   and expose it both as `lastMessage` and through a per-type
  11:  *   subscription API (`subscribe("avatar.directive", handler)`).
  12:  * - Provide `send(msg: ClientMessage)` that serialises to JSON.
  13:  *
  14:  * The factory is pluggable so tests can supply a `MockWebSocket`.
  15:  */
  16: 
  17: import {
  18:   createContext,
  19:   useCallback,
  20:   useContext,
  21:   useEffect,
  22:   useMemo,
  23:   useRef,
  24:   useState,
  25:   type PropsWithChildren,
  26: } from "react";
  27: 
  28: import { createReconnectController, type ReconnectConfig } from "./reconnect";
  29: import {
  30:   isServerMessage,
  31:   type ClientMessage,
  32:   type ServerMessage,
  33:   type ServerMessageType,
  34: } from "./protocol";

  39: // ---------------------------------------------------------------------------

  41: export type ConnectionStatus = "idle" | "connecting" | "open" | "closed";
  42:

  44:   msg: Extract<ServerMessage, { type: T }>,

  48:   status: ConnectionStatus;
  49:   lastMessage: ServerMessage | null;

  92: function defaultUrl(): string {
  93:   if (typeof window === "undefined") return "ws://localhost:8000/ws";
  94:   const configured = import.meta.env["VITE_OPENMIMICRY_WS_URL"];
  95:   if (configured) return configured;
  96:   if (
  97:     "__TAURI_INTERNALS__" in window ||
  98:     "__TAURI_IPC__" in window ||
  99:     window.location.port === "5173"
 100:   ) {
 101:     return "ws://127.0.0.1:8000/ws";
 102:   }
 103:   const proto = window.location.protocol === "https:" ? "wss" : "ws";
 104:   return `${proto}://${window.location.host}/ws`;

 107: function defaultFactory(url: string): WebSocketLike {
 108:   // The browser global has a wider type than our shrunk-down interface,
 109:   // but every method we use is present on it.
 110:   return new WebSocket(url) as unknown as WebSocketLike;
 111: }
 112: 
 113: export function WSProvider(props: WSProviderProps): JSX.Element {
 114:   const {
 115:     url = defaultUrl(),
 116:     socketFactory = defaultFactory,
 117:     reconnect: reconnectCfg,
 118:     autoConnect = true,
 119:     children,
 120:   } = props;
 121: 
 122:   const [status, setStatus] = useState<ConnectionStatus>("idle");
 123:   const [lastMessage, setLastMessage] = useState<ServerMessage | null>(null);
 124: 
 125:   // The current socket is mutable across renders; the reconnect loop
 126:   // closes around `socketRef.current`.
 127:   const socketRef = useRef<WebSocketLike | null>(null);
 128:   const listenersRef = useRef<Map<ServerMessageType, Set<Listener<ServerMessageType>>>>(
 129:     new Map(),
 130:   );
 131:   const ctrlRef = useRef(createReconnectController(reconnectCfg));
 132:   const stoppedRef = useRef(false);
 133:   const connectionGenerationRef = useRef(0);
 134:   const forceReconnectRef = useRef<(() => void) | null>(null);
 135: 
 136:   // -----------------------------------------------------------------------
 137:   // Connection lifecycle
 138:   // -----------------------------------------------------------------------
 139: 
 140:   const connect = useCallback(() => {
 141:     if (stoppedRef.current) return;
 142:     setStatus("connecting");
 143:     const ws = socketFactory(url);
 144:     const generation = ++connectionGenerationRef.current;
 145:     socketRef.current = ws;
 146:     const isCurrent = (): boolean =>
 147:       socketRef.current === ws && connectionGenerationRef.current === generation;
 148: 
 149:     const onOpen = (): void => {
 150:       if (!isCurrent()) return;
 151:       ctrlRef.current.reset();
 152:       setStatus("open");
 153:     };
 154: 
 155:     const onMessage = (ev: Event): void => {
 156:       if (!isCurrent()) return;
 157:       const data = (ev as unknown as { data?: string }).data;
 158:       if (typeof data !== "string") return;
 159: 
 160:       let parsed: unknown;
 161:       try {
 162:         parsed = JSON.parse(data);
 163:       } catch {
 164:         return;
 165:       }
 166: 
 167:       if (!isServerMessage(parsed)) return;
 168: 
 169:       const msg = parsed;
 170:       setLastMessage(msg);
 171: 
 172:       const bucket = listenersRef.current.get(msg.type);
 173:       if (!bucket) return;
 174: 
 175:       for (const fn of Array.from(bucket)) {
 176:         try {
 177:           (fn as Listener<typeof msg.type>)(
 178:             msg as Extract<ServerMessage, { type: typeof msg.type }>,
 179:           );
 180:         } catch {
 181:           // listener errors are swallowed; one broken consumer doesn't
 182:           // poison the dispatch loop.
 183:         }
 184:       }
 185:     };
 186: 
 187:     const onClose = (): void => {
 188:       // React StrictMode mounts, cleans up, then mounts again in development.
 189:       // A delayed close/message from that first socket must never clear or
 190:       // reconnect the active second socket.
 191:       if (!isCurrent()) return;
 192:       setStatus("closed");
 193:       socketRef.current = null;
 194:       if (stoppedRef.current) return;
 195:       void scheduleReconnect();
 196:     };
 197: 
 198:     const onError = (): void => {
 199:       if (!isCurrent()) return;
 200:       // Errors land as `close` on most browsers; some send `error`
 201:       // separately. We rely on `close` to drive reconnect.
 202:     };
 203: 
 204:     ws.addEventListener("open", onOpen);
 205:     ws.addEventListener("message", onMessage);
 206:     ws.addEventListener("close", onClose);
 207:     ws.addEventListener("error", onError);
