   1: """Concrete ``EventBus`` — in-process pub/sub with bounded per-subscriber queues.
   2: 
   3: The shape of this class is frozen by ``openmimicry.core.contracts.bus``. This
   4: module is the M0-shipped implementation. The single instance is owned by
   5: ``Runtime``; modules receive it as a dependency rather than reaching for a
   6: process-global singleton.
   7: 
   8: Drop policy
   9: -----------
  10: 
  11: Each subscriber owns its own ``asyncio.Queue`` of ``maxsize`` (default 1024).
  12: ``publish`` is non-blocking; if a subscriber's queue is full the **oldest**
  13: event is dropped to make room for the newest, and a single warning per
  14: subscriber is logged. This favours liveness of the producer (typically the
  15: adapter thread) over delivery guarantees on a stalled consumer.
  16: 
  17: If you need lossless delivery, drain ``subscribe()`` faster than the producer.
  18: """
  19: 
  20: from __future__ import annotations
  21: 
  22: import asyncio
  23: import contextlib
  24: import logging
  25: from collections.abc import AsyncIterator
  26: 
  27: from .schemas.events import RuntimeEvent
  28: 
  29: __all__ = ["DEFAULT_QUEUE_MAXSIZE", "EventBus"]
  30: 
  31: 
  32: DEFAULT_QUEUE_MAXSIZE: int = 1024
  33: 
  34: 
  35: _log = logging.getLogger(__name__)
  36: 
  37: 
  38: class _Subscription:
  39:     """One live subscription. Owns its queue and a one-shot warn flag."""
  40: 
  41:     __slots__ = ("_closed", "_warned", "queue")
  42: 
  43:     def __init__(self, maxsize: int) -> None:
  44:         self.queue: asyncio.Queue[RuntimeEvent | None] = asyncio.Queue(maxsize=maxsize)
  45:         self._warned: bool = False
  46:         self._closed: bool = False
  47: 
  48:     @property
  49:     def closed(self) -> bool:
  50:         return self._closed
  51: 
  52:     def offer(self, event: RuntimeEvent) -> None:
  53:         """Non-blocking put. Drops the oldest event if the queue is full."""
  54:         if self._closed:
  55:             return
  56:         try:
  57:             self.queue.put_nowait(event)
  58:         except asyncio.QueueFull:
  59:             with contextlib.suppress(asyncio.QueueEmpty):
  60:                 self.queue.get_nowait()
  61:             with contextlib.suppress(asyncio.QueueFull):
  62:                 self.queue.put_nowait(event)
  63:             if not self._warned:
  64:                 self._warned = True
  65:                 _log.warning(
  66:                     "EventBus subscriber fell behind; dropping oldest events "
  67:                     "(maxsize=%d). Drain subscribe() faster or raise maxsize.",
  68:                     self.queue.maxsize,
  69:                 )
  70: 
  71:     def close(self) -> None:
  72:         """Send a sentinel so a pending ``await get()`` wakes up and stops."""
  73:         if self._closed:
  74:             return
  75:         self._closed = True
  76:         try:
  77:             self.queue.put_nowait(None)
  78:         except asyncio.QueueFull:
  79:             with contextlib.suppress(asyncio.QueueEmpty):
  80:                 self.queue.get_nowait()
  81:             with contextlib.suppress(asyncio.QueueFull):
  82:                 self.queue.put_nowait(None)

  84: 
  85: class EventBus:
  86:     """The concrete in-process event bus."""
  87: 
  88:     def __init__(self, *, maxsize: int = DEFAULT_QUEUE_MAXSIZE) -> None:
  89:         self._maxsize = maxsize
  90:         self._subs: list[_Subscription] = []
  91:         self._closed: bool = False
  92: 
  93:     def publish(self, event: RuntimeEvent) -> None:
  94:         """Fan out one event to every active subscriber. Non-blocking."""
  95:         if self._closed:
  96:             return
  97:         self._reap_closed()
  98:         for sub in self._subs:
  99:             sub.offer(event)
 100: 
 101:     def subscribe(self) -> AsyncIterator[RuntimeEvent]:
 102:         """Open a new subscription. Closes when the bus closes.
 103: 
 104:         Registration is synchronous at call time so any ``publish`` happening
 105:         before the caller starts iterating still queues for this subscriber.
 106:         The returned async iterator drains that queue until the sentinel.
 107:         """
 108:         self._reap_closed()
 109:         sub = _Subscription(maxsize=self._maxsize)
 110:         if self._closed:
 111:             sub.close()
 112:         else:
 113:             self._subs.append(sub)
 114:         return self._iter(sub)
 115: 
 116:     async def _iter(self, sub: _Subscription) -> AsyncIterator[RuntimeEvent]:
 117:         """Drain ``sub`` until the bus or the subscriber closes."""
 118:         try:
 119:             while True:
 120:                 item = await sub.queue.get()
 121:                 if item is None:
 122:                     return
 123:                 yield item
 124:         finally:
 125:             sub.close()
 126:             with contextlib.suppress(ValueError):
 127:                 self._subs.remove(sub)
 128: 
 129:     async def aclose(self) -> None:
 130:         """Drain and close every subscriber. Safe to call multiple times."""
 131:         if self._closed:
 132:             return
 133:         self._closed = True
 134:         for sub in list(self._subs):
 135:             sub.close()
 136:         # Drop all references so subscriber_count reflects closed state even
 137:         # for subscribers whose async-generator finally hasn't run yet.
 138:         self._subs.clear()
 139:         await asyncio.sleep(0)
 140: 
 141:     def _reap_closed(self) -> None:
 142:         """Drop any subscriptions whose underlying queue has been closed."""
 143:         self._subs[:] = [s for s in self._subs if not s.closed]
 144: 
 145:     @property
 146:     def subscriber_count(self) -> int:
 147:         """The number of currently-open subscribers. Mostly useful in tests."""
 148:         self._reap_closed()
 149:         return len(self._subs)

 151:     @property
 152:     def closed(self) -> bool:
 153:         return self._closed
