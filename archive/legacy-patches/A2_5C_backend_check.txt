  92:     ) -> None:
  93:         text = await _collect_text(text_or_stream)
  94:         if not text.strip():
  95:             return
  96:         if len(text) > 10_000:
  97:             raise ValueError("ElevenLabs multilingual v2 input exceeds 10,000 characters")
  98:         await self.stop()
  99:         self._ready = asyncio.Event()
 100:         self._is_speaking = True
 101:         try:
 102:             pcm = await asyncio.to_thread(self._synthesize, text)
 103:             self._mouth_frames = pcm_to_mouth_frames(pcm)
 104:             if on_chunk is not None:
 105:                 on_chunk({
 106:                     "type": "lip_sync.frames",
 107:                     "window_ms": 40,
 108:                     "frames": list(self._mouth_frames),

 110:             with tempfile.TemporaryDirectory(prefix="openmimicry-azure-") as tmp:
 111:                 wave_path = Path(tmp) / "utterance.wav"
 112:                 _write_pcm_wave(wave_path, pcm, sample_rate=24000)
 113:                 command, environment = _playback_command(wave_path)
 114:                 self._process = await asyncio.create_subprocess_exec(
 115:                     *command,
 116:                     env=environment,
 117:                     stdout=asyncio.subprocess.DEVNULL,
 118:                     stderr=asyncio.subprocess.PIPE,
 119:                 )
 120:                 self._ready.set()
 121:                 _stdout, stderr = await self._process.communicate()
 122:                 if self._process.returncode not in {0, None}:
 123:                     raise RuntimeError(
 124:                         "Azure Speech audio playback failed: "
 125:                         + stderr.decode("utf-8", errors="replace")[-1000:]

 127:         finally:
 128:             self._process = None
 129:             self._is_speaking = False

 131:     def _synthesize(self, text: str) -> bytes:
 132:         from xml.sax.saxutils import escape
 133: 
 134:         voice = escape(self._settings.voice, {'"': '&quot;'})
 135:         ssml = (
 136:             '<speak version="1.0" '
 137:             'xmlns="http://www.w3.org/2001/10/synthesis" '
 138:             'xml:lang="zh-HK">'
 139:             f'<voice name="{voice}">{escape(text)}</voice>'
 140:             '</speak>'
 141:         )
 142:         request = Request(
 143:             self._settings.endpoint,
 144:             data=ssml.encode("utf-8"),
 145:             headers={
 146:                 "Ocp-Apim-Subscription-Key": self._api_key(),
 147:                 "Content-Type": "application/ssml+xml",
 148:                 "X-Microsoft-OutputFormat": "raw-24khz-16bit-mono-pcm",
 149:                 "User-Agent": "OpenMimicry/1.6",
 150:             },
 151:             method="POST",
 152:         )
 153:         try:
 154:             with urlopen(request, timeout=90.0) as response:
 155:                 audio = response.read(_MAX_AUDIO_BYTES + 1)
 156:         except HTTPError as exc:
 157:             detail = exc.read(2048).decode("utf-8", errors="replace")
 158:             raise RuntimeError(f"Azure Speech returned HTTP {exc.code}: {detail}") from exc
 159:         except (URLError, TimeoutError, OSError) as exc:
 160:             raise RuntimeError(f"Azure Speech request failed: {exc}") from exc
 161:         if len(audio) > _MAX_AUDIO_BYTES:
 162:             raise RuntimeError("Azure Speech audio response exceeded 64 MiB")
 163:         if not audio:
 164:             raise RuntimeError("Azure Speech returned an empty audio response")
 165:         return audio

  19: class AzureQwen3FallbackTTSAdapter:
  20:     name = "azure-qwen3-fallback"
  21: 
  22:     def __init__(
  23:         self,
  24:         azure: AzureTTSAdapter,
  25:         qwen3: Qwen3TTSAdapter,
  26:     ) -> None:
  27:         self._azure = azure
  28:         self._qwen3 = qwen3
  29:         self._active = None
  30:         self._ready = asyncio.Event()
  31:         self._speaking = False
  32:         self._generation = 0
  33: 
  34:     @property
  35:     def is_speaking(self) -> bool:
  36:         return self._speaking
  37: 
  38:     async def prepare(self, config: TTSConfig) -> None:
  39:         # Do not preload Qwen3 while Azure is available.
  40:         try:
  41:             await self._azure.prepare(config)
  42:         except Exception as exc:
  43:             logger.warning(
  44:                 "Azure TTS preparation failed; Qwen3 remains available as fallback: %s",
  45:                 exc,
  46:             )
  47: 
  48:     async def speak(
  49:         self,
  50:         text_or_stream: str | AsyncIterable[str],
  51:         *,
  52:         config: TTSConfig,
  53:         on_chunk=None,
  54:     ) -> None:
  55:         text = await _collect_text(text_or_stream)
  56:         if not text.strip():
  57:             return
  58: 
  59:         self._generation += 1
  60:         generation = self._generation
  61:         self._ready.clear()
  62: 
  63:         self._speaking = True
  64:         self._active = self._azure
  65: 
  66:         async def play(adapter) -> None:
  67:             task = asyncio.create_task(
  68:                 adapter.speak(text, config=config, on_chunk=on_chunk)
  69:             )
  70:             waiter = getattr(adapter, "wait_until_ready", None)
  71:             ready_task = (
  72:                 asyncio.create_task(waiter(timeout_s=180.0))
  73:                 if callable(waiter)
  74:                 else None
  75:             )
  76:             try:
  77:                 if ready_task is not None:
  78:                     done, _ = await asyncio.wait(
  79:                         {task, ready_task},
  80:                         return_when=asyncio.FIRST_COMPLETED,
  81:                     )
  82:                     if task in done:
  83:                         await task
  84:                         if generation == self._generation:
  85:                             self._ready.set()
  86:                     if ready_task in done and ready_task.result():
  87:                         if generation == self._generation:
  88:                             self._ready.set()
  89:                 else:
  90:                     if generation == self._generation:
  91:                         self._ready.set()
  92:                 await task
  93:             finally:
  94:                 if ready_task is not None:
  95:                     if not ready_task.done():
  96:                         ready_task.cancel()
  97:                     with suppress(asyncio.CancelledError):
  98:                         await ready_task
  99:                 if not task.done():
 100:                     task.cancel()
 101:                     with suppress(asyncio.CancelledError):
 102:                         await task
 103: 
 104:         try:
 105:             try:
 106:                 await play(self._azure)
 107:             except asyncio.CancelledError:
 108:                 raise
 109:             except Exception as exc:
 110:                 logger.warning(
 111:                     "Azure TTS failed; switching to Qwen3: %s", exc
 112:                 )
 113:                 if generation != self._generation:
 114:                     return
 115:                 self._active = self._qwen3
 116:                 await self._qwen3.prepare(config)
 117:                 await play(self._qwen3)
 118:         finally:
 119:             if generation == self._generation:
 120:                 self._speaking = False
 121:                 self._active = None
 122: 
 123:     async def wait_until_ready(
 124:         self, *, timeout_s: float = 30.0
 125:     ) -> bool:
 126:         if not self._speaking and self._ready.is_set():
 127:             self._ready.clear()
 128:         try:
 129:             async with asyncio.timeout(timeout_s):
 130:                 await self._ready.wait()
 131:             return True
 132:         except TimeoutError:
 133:             return False
 134: 
 135:     async def stop(self) -> None:
 136:         self._generation += 1
 137:         self._speaking = False
 138:         adapter = self._active
 139:         self._active = None
 140:         if adapter is not None:
 141:             await adapter.stop()
 142: 
 143:     async def healthcheck(self) -> bool:
 144:         return await self._azure.healthcheck()
 145: 
 146:     async def close(self) -> None:
 147:         await self.stop()
 148:         await self._azure.close()
 149:         await self._qwen3.close()

 260:         utterance_id = utterance_id or uuid4().hex
 261:         ready = asyncio.get_running_loop().create_future()
 262:         self._current_utterance_id = utterance_id
 263:         self._speech_ready = ready
 264:         self._bus.publish(TTSQueued(ts=_now(), utterance_id=utterance_id))
 265:         self._current_tts_task = asyncio.create_task(
 266:             self._speak_once(text_or_stream, utterance_id, ready, resume),
 267:             name=f"openmimicry.voice.say.{utterance_id[:8]}",
 268:         )
 269:         return utterance_id

 271:     async def wait_until_speech_ready(
 272:         self, *, timeout_s: float = 10.0, utterance_id: str | None = None
 273:     ) -> bool:
 274:         """Wait until the TTS adapter reports that speaker playback began."""
 275: 
 276:         if utterance_id is not None and utterance_id != self._current_utterance_id:
 277:             return False
 278:         ready = self._speech_ready
 279:         if ready is None:
 280:             return False
 281:         try:
 282:             async with asyncio.timeout(max(0.05, timeout_s)):
 283:                 return bool(await asyncio.shield(ready))
 284:         except TimeoutError:
 285:             _log.error("SpeechController: TTS readiness timed out after %.1fs", timeout_s)
 286:             return False
 287:         except Exception as exc:
 288:             _log.warning("SpeechController: TTS readiness wait failed: %s", exc)
 289:             return False

 291:     async def _speak_once(
 292:         self,
 293:         text_or_stream: str | AsyncIterable[str],
 294:         utterance_id: str,
 295:         ready_future: asyncio.Future[bool],
 296:         resume_listening: tuple[str, list[str]] | None,
 297:     ) -> None:
 298:         tts_config = TTSConfig(
 299:             engine=self._cfg.tts.engine,
 300:             voice=self._cfg.tts.voice,
 301:             rate=self._cfg.tts.rate,
 302:             interruptible=self._cfg.tts.interruptible,
 303:         )
 304:         cancelled = False
 305:         failure_message: str | None = None
 306:         play_task: asyncio.Task[None] | None = None
 307:         controller_task = asyncio.current_task()
 308: 
 309:         def _publish_lip_sync_frames(chunk: object) -> None:
 310:             if self._current_utterance_id != utterance_id:
 311:                 return
 312:             if (
 313:                 self._current_tts_task is not controller_task
 314:                 or controller_task is None
 315:                 or controller_task.done()
 316:             ):
 317:                 return
 318:             if not isinstance(chunk, dict):
 319:                 return
 320:             if chunk.get("type") != "lip_sync.frames":
 321:                 return
 322: 
 323:             frames = chunk.get("frames")
 324:             window_ms = chunk.get("window_ms", 40)
 325: 
 326:             if not isinstance(frames, list):
 327:                 return
 328: 
 329:             try:
 330:                 event = TTSLipSyncFrames(
 331:                     ts=_now(),
 332:                     utterance_id=utterance_id,
 333:                     window_ms=window_ms,
 334:                     frames=frames,
 335:                 )
 336:             except (TypeError, ValueError) as exc:
 337:                 _log.warning("Invalid lip-sync frames: %s", exc)
 338:                 return
 339: 
 340:             self._bus.publish(event)
 341: 
 342:         try:
 343:             play_task = asyncio.create_task(
 344:                 self._tts.speak(
 345:                     text_or_stream,
 346:                     config=tts_config,
 347:                     on_chunk=_publish_lip_sync_frames,
 348:                 ),
 349:                 name="openmimicry.voice.tts_adapter",
 350:             )
 351:             # Give the adapter one loop turn to replace its prior readiness
 352:             # Event. Without this, reply two could observe reply one's Event.
 353:             await asyncio.sleep(0)
 354:             waiter = getattr(self._tts, "wait_until_ready", None)
 355:             if callable(waiter):
 356:                 ready_task = asyncio.create_task(
 357:                     cast(Callable[..., Awaitable[Any]], waiter)(
 358:                         timeout_s=self._cfg.tts.readiness_timeout_s
 359:                     ),
 360:                     name="openmimicry.voice.tts_ready",
 361:                 )
 362:                 done, _pending = await asyncio.wait(
 363:                     {ready_task, play_task}, return_when=asyncio.FIRST_COMPLETED
 364:                 )
 365:                 audio_ready = bool(ready_task.result()) if ready_task in done else False
 366:                 if not ready_task.done():
 367:                     ready_task.cancel()
 368:                     with suppress(asyncio.CancelledError, Exception):
 369:                         await ready_task
 370:             else:
 371:                 audio_ready = True
 372:             self._resolve_speech_ready(ready_future, audio_ready)
 373:             # Speaking is a playback state, not a synthesis/queue state.  A
 374:             # failed or stale TTS job must never replace listening/thinking.
 375:             if audio_ready and self._current_tts_task is asyncio.current_task():
 376:                 self._tts_ready = True
 377:                 self._bus.publish(TTSReady(ts=_now(), utterance_id=utterance_id))
 378:                 self._bus.publish(TTSStarted(ts=_now(), utterance_id=utterance_id))
 379:             elif not audio_ready:
 380:                 failure_message = "Text-to-speech playback did not become ready."
 381:             timeout_s = _playback_timeout_seconds(text_or_stream)
 382:             try:
 383:                 async with asyncio.timeout(timeout_s):
 384:                     await play_task
 385:             except TimeoutError:
 386:                 _log.error(
 387:                     "SpeechController: TTS playback exceeded %.1fs; forcing recovery",
 388:                     timeout_s,
 389:                 )
 390:                 failure_message = f"Text-to-speech playback exceeded {timeout_s:.1f}s."
 391:                 if not play_task.done():
 392:                     play_task.cancel()
 393:                 with suppress(Exception):
 394:                     await self._tts.stop()
 395:                 with suppress(asyncio.CancelledError, Exception):
 396:                     await play_task
 397:         except asyncio.CancelledError:
 398:             cancelled = True
 399:             if play_task is not None and not play_task.done():
 400:                 play_task.cancel()
 401:             with suppress(Exception):
 402:                 await self._tts.stop()
 403:             if play_task is not None:
 404:                 with suppress(asyncio.CancelledError, Exception):
 405:                     await play_task
 406:             self._resolve_speech_ready(ready_future, False)
 407:             raise
 408:         except Exception as exc:
 409:             self._tts_ready = False
 410:             failure_message = str(exc) or type(exc).__name__
 411:             _log.warning("SpeechController: tts.speak raised: %s", exc, exc_info=True)
 412:             if play_task is not None and not play_task.done():
 413:                 play_task.cancel()
 414:             with suppress(Exception):
 415:                 await self._tts.stop()
 416:             if play_task is not None:
 417:                 with suppress(asyncio.CancelledError, Exception):
 418:                     await play_task
 419:             self._resolve_speech_ready(ready_future, False)
 420:         finally:
 421:             self._resolve_speech_ready(ready_future, False)
 422:             if cancelled:
 423:                 self._bus.publish(TTSInterrupted(ts=_now(), utterance_id=utterance_id))
 424:             elif failure_message is not None:
 425:                 self._bus.publish(
 426:                     TTSFailed(
 427:                         ts=_now(),
 428:                         utterance_id=utterance_id,
 429:                         message=failure_message,
 430:                     )
 431:                 )
 432:             else:
 433:                 self._bus.publish(TTSFinished(ts=_now(), utterance_id=utterance_id))
 434:             if self._current_utterance_id == utterance_id:
 435:                 self._current_utterance_id = None
 436:                 self._current_tts_task = None
 437:                 self._speech_ready = None
 438:             if resume_listening is not None and self._started and not self._ptt_active:
 439:                 try:
 440:                     await self._restore_passive_listening(resume_listening)
 441:                 except Exception as exc:
 442:                     _log.warning(
 443:                         "SpeechController: could not restore listening after TTS: %s",
 444:                         exc,
 445:                         exc_info=True,
 446:                     )
 447: 
 448:     @staticmethod
 449:     def _resolve_speech_ready(future: asyncio.Future[bool], ready: bool) -> None:
 450:         if future is not None and not future.done():
 451:             future.set_result(ready)

 453:     async def interrupt(self) -> None:
 454:         """Request TTS stop without letting a device driver freeze input.
 455: 
 456:         Third-party audio shutdown continues in the background when it cannot
 457:         settle within the short PTT latency budget.  The adapter owns the
 458:         bounded worker cleanup and replacement lane.
 459:         """
 460: 
 461:         task = self._current_tts_task
 462:         self._current_tts_task = None
 463:         self._current_utterance_id = None
 464:         if task is None or task.done():
 465:             # Nothing is playing; keep the persistent TTS worker alive.
 466:             return
 467:         task.cancel()
 468:         try:
 469:             async with asyncio.timeout(_INTERRUPT_SETTLE_TIMEOUT_SECONDS):
 470:                 await asyncio.shield(task)
 471:         except TimeoutError:
 472:             _log.warning(
 473:                 "SpeechController: TTS interrupt still settling after %.2fs; "
 474:                 "voice input will continue while audio cleanup runs",
 475:                 _INTERRUPT_SETTLE_TIMEOUT_SECONDS,
 476:             )
 477:             task.add_done_callback(_consume_background_task)
 478:         except (asyncio.CancelledError, Exception):
 479:             pass

 481:     async def _barge_in_loop(self) -> None:
 482:         """Watch ``stt.vad_active``; interrupt TTS on persisting activity."""
 483:         grace_s = max(0.0, self._cfg.modes.barge_in_grace_ms / 1000.0)
 484:         poll_s = max(0.02, grace_s / 4 if grace_s else 0.05)
 485:         while True:
 486:             try:
 487:                 await asyncio.sleep(poll_s)
 488:                 if not self._cfg.modes.barge_in_enabled:
 489:                     continue
 490:                 if not self._cfg.tts.interruptible:
 491:                     continue
 492:                 if not self._tts.is_speaking:
 493:                     continue
 494:                 if not self._stt.vad_active:
 495:                     continue
 496:                 # Grace period: wait, then re-check.
 497:                 if grace_s:
 498:                     await asyncio.sleep(grace_s)
 499:                 if not self._tts.is_speaking:
 500:                     continue
 501:                 if not self._stt.vad_active:
 502:                     continue
 503:                 _log.info("SpeechController: barge-in; interrupting TTS")
 504:                 await self.interrupt()
 505:             except asyncio.CancelledError:
 506:                 raise
 507:             except Exception as exc:
 508:                 _log.warning("SpeechController.barge_in_loop: %s", exc, exc_info=True)

 512:     async def ptt_down(self) -> None:
 513:         """Open dictation STT. Stops any in-flight TTS first."""
 514:         if self._ptt_active:
 515:             return
 516:         await self.interrupt()
 517:         resume: tuple[str, list[str]] | None = None
 518:         if self._live_listening and self._listening_mode is not None:
 519:             resume = (self._listening_mode, list(self._live_wake_names))
 520:             await self._stop_passive_listening()
 521:         try:
 522:             await self._stt.start(self._dictation_config(mode="push_to_talk"))
 523:             self._stt_ready = True
 524:         except Exception:
 525:             self._stt_ready = False
 526:             if resume is not None:
 527:                 await self._restore_passive_listening(resume)
 528:             raise
 529:         self._resume_listening_after_ptt = resume
 530:         self._ptt_active = True
 531:         self._bus.publish(UserSpeechStarted(ts=_now()))
 532: 
 533:     async def ptt_up(self) -> None:
 534:         """Close dictation STT; publish the next final transcript as the result."""
 535:         if not self._ptt_active:
 536:             return
 537:         # Read the next final transcript (with a short timeout so a noisy
 538:         # tail doesn't deadlock the caller).
 539:         resume = self._resume_listening_after_ptt
 540:         self._resume_listening_after_ptt = None
 541:         try:
 542:             # First-run Whisper model initialization and ordinary Windows CPU
 543:             # transcription can exceed two seconds after the button is
 544:             # released. Keep the microphone turn alive long enough to receive
 545:             # the final transcript instead of silently submitting no speech.
 546:             finish_utterance = getattr(self._stt, "finish_utterance", None)
 547:             if callable(finish_utterance):
 548:                 await cast(Callable[..., Awaitable[Any]], finish_utterance)()
 549:             text, reason = await self._await_final_transcript(timeout_s=_PTT_FINAL_TIMEOUT_SECONDS)
 550:         except Exception:
 551:             with suppress(Exception):
 552:                 await self._stt.stop()
 553:             self._ptt_active = False
 554:             if resume is not None:
 555:                 await self._restore_passive_listening(resume)
 556:             raise
 557:         with suppress(Exception):
 558:             await self._stt.stop()
 559:         self._ptt_active = False
 560:         _log.info("SpeechController: PTT transcript (%s): %r", reason, text)
 561:         self._bus.publish(
 562:             UserSpeechFinal(
 563:                 ts=_now(),
 564:                 text=text,
 565:                 raw_text=text,
 566:                 reason=reason,
 567:                 accepted=bool(text) and reason == "normal",
 568:                 input_mode="push_to_talk",
 569:                 rejection_reason=None if text else "empty",
 570:             )
 571:         )
 572:         if resume is not None:
 573:             await self._restore_passive_listening(resume)

 575:     async def _await_final_transcript(self, *, timeout_s: float) -> tuple[str, str]:
 576:         """Drain the STT stream until a final transcript arrives or we time out.
 577: 
 578:         Returns ``(text, reason)`` matching ``UserSpeechFinal``.
 579:         """
 580:         last_partial = ""
 581:         try:
 582:             async with asyncio.timeout(timeout_s):
 583:                 async for transcript in self._stt.transcripts:
 584:                     if transcript.is_final:
 585:                         return transcript.text, "normal"
 586:                     last_partial = transcript.text
 587:                     self._bus.publish(
 588:                         TranscriptPreview(
 589:                             ts=_now(),
 590:                             text=transcript.text,
 591:                             is_final=False,
 592:                         )
 593:                     )
 594:         except TimeoutError:
 595:             return last_partial, "no_speech" if not last_partial else "normal"
 596:         return last_partial, "no_speech" if not last_partial else "normal"

 718:     async def set_post_speech_silence_duration(self, seconds: float) -> None:
 719:         """Update phrase endpointing and restart an active passive listener.
 720: 
 721:         Push-to-talk reads the new value the next time its microphone session
 722:         starts. Continuous and wake listeners are restarted immediately so the
 723:         dashboard setting has an observable, deterministic effect.
 724:         """
 725: 
 726:         value = float(seconds)
 727:         if not 0.2 <= value <= 3.0:
 728:             raise ValueError("end-of-speech pause must be between 0.2 and 3.0 seconds")
 729:         if value == self._cfg.stt.post_speech_silence_duration:
 730:             return
 731: 
 732:         passive_state: tuple[str, list[str]] | None = None
 733:         if self._live_listening and self._listening_mode is not None:
 734:             passive_state = (self._listening_mode, list(self._live_wake_names))
 735:             await self._stop_passive_listening()
 736: 
 737:         stt_config = self._cfg.stt.model_copy(update={"post_speech_silence_duration": value})
 738:         self._cfg = self._cfg.model_copy(update={"stt": stt_config})
 739: 
 740:         if passive_state is not None:
 741:             await self._restore_passive_listening(passive_state)

 789:     async def _live_listener(self) -> None:
 790:         """Project transcripts while continuous or wake listening is active."""
 791:         wake_announced = False
 792:         _log.info("SpeechController: %s listener consuming transcripts", self._listening_mode)
 793:         try:
 794:             async for transcript in self._stt.transcripts:
 795:                 if transcript.is_final:
 796:                     _log.info(
 797:                         "SpeechController: %s final dequeued: chars=%d",
 798:                         self._listening_mode,
 799:                         len(transcript.text),
 800:                     )
 801:                 wake_match: tuple[str, str] | None = None
 802:                 if self._listening_mode == "wake":
 803:                     wake_match = _extract_wake_command(transcript.text, self._live_wake_names)
 804:                     if wake_match is None:
 805:                         if transcript.is_final:
 806:                             wake_announced = False
 807:                             raw = transcript.text.strip()
 808:                             if raw:
 809:                                 self._bus.publish(
 810:                                     UserSpeechFinal(
 811:                                         ts=_now(),
 812:                                         text=raw,
 813:                                         raw_text=raw,
 814:                                         accepted=False,
 815:                                         input_mode="wake",
 816:                                         rejection_reason="wake_name_missing",
 817:                                     )
 818:                                 )
 819:                         continue
 820:                     wake_name, command = wake_match
 821:                     if not wake_announced:
 822:                         self._bus.publish(WakeDetected(ts=_now(), name=wake_name))
 823:                         wake_announced = True
 824:                 else:
 825:                     command = transcript.text.strip()
 826: 
 827:                 if transcript.is_final:
 828:                     wake_announced = False
 829:                     raw = transcript.text.strip()
 830:                     if command:
 831:                         accepted = True
 832:                         rejection_reason = None
 833:                         if self._listening_mode == "wake" and self._is_duplicate_wake(command):
 834:                             accepted = False
 835:                             rejection_reason = "duplicate"
 836:                         _log.info(
 837:                             "SpeechController: %s transcript %s: %r",
 838:                             self._listening_mode,
 839:                             "accepted" if accepted else "ignored duplicate",
 840:                             command,
 841:                         )
 842:                         self._bus.publish(
 843:                             UserSpeechFinal(
 844:                                 ts=_now(),
 845:                                 text=command,
 846:                                 raw_text=raw,
 847:                                 reason="normal",
 848:                                 accepted=accepted,
 849:                                 input_mode=(
 850:                                     "wake" if self._listening_mode == "wake" else "continuous"
 851:                                 ),
 852:                                 rejection_reason=rejection_reason,
 853:                             )
 854:                         )
 855:                     elif raw:
 856:                         self._bus.publish(
 857:                             UserSpeechFinal(
 858:                                 ts=_now(),
 859:                                 text=raw,
 860:                                 raw_text=raw,
 861:                                 accepted=False,
 862:                                 input_mode="wake",
 863:                                 rejection_reason="empty",
 864:                             )
 865:                         )
 866:                 else:
 867:                     if command:
 868:                         self._bus.publish(
 869:                             TranscriptPreview(ts=_now(), text=command, is_final=False)
 870:                         )
 871:         except asyncio.CancelledError:
 872:             raise
 873:         except Exception as exc:
 874:             _log.warning("SpeechController._live_listener: %s", exc, exc_info=True)
 875:         finally:
 876:             _log.info("SpeechController: passive listener stopped")

 878:     def _is_duplicate_wake(self, command: str) -> bool:
 879:         normalised = " ".join(command.casefold().split())
 880:         now = time.monotonic()
 881:         previous = self._last_accepted_wake
 882:         if (
 883:             previous is not None
 884:             and previous[0] == normalised
 885:             and now - previous[1] <= _WAKE_DUPLICATE_WINDOW_SECONDS
 886:         ):
 887:             return True
 888:         self._last_accepted_wake = (normalised, now)
 889:         return False

 892: def _normalise_wake_names(names: list[str]) -> list[str]:
 893:     """Trim, de-duplicate, and prefer the longest prefix during matching."""
 894: 
 895:     unique: dict[str, str] = {}
 896:     for raw in names:
 897:         name = " ".join(str(raw).split()).strip(" ,.:;!?-")
 898:         if name:
 899:             unique.setdefault(name.casefold(), name)
 900:     return sorted(unique.values(), key=len, reverse=True)

 903: def _playback_timeout_seconds(text_or_stream: str | AsyncIterable[str]) -> float:
 904:     """Bound a broken audio stream without truncating normal long replies."""
 905: 
 906:     if not isinstance(text_or_stream, str):
 907:         return _TTS_MAX_TIMEOUT_SECONDS
 908:     # A conservative 8 characters/second plus startup/driver padding.
 909:     estimate = 12.0 + len(text_or_stream) / 8.0
 910:     return min(_TTS_MAX_TIMEOUT_SECONDS, max(_TTS_MIN_TIMEOUT_SECONDS, estimate))

 913: def _extract_wake_command(text: str, names: list[str]) -> tuple[str, str] | None:
 914:     """Return ``(matched_name, command)`` only when ``text`` begins with a name."""
 915: 
 916:     for name in names:
 917:         match = re.match(
 918:             rf"^\s*{re.escape(name)}(?=$|[\s,.:;!?-])",
 919:             text,
 920:             flags=re.IGNORECASE,
 921:         )
 922:         if match is None:
 923:             continue
 924:         command = text[match.end() :].lstrip(" \t,.:;!?-")
 925:         return name, command
 926:     return None

   2: 
   3: The wire protocol is defined in ``docs/contracts.md`` §9. There are six
   4: inbound message shapes (server -> frontend):
   5: 
   6: * ``avatar.directive`` — emitted by the avatar runtime via its WSBridge,
   7:   **not** synthesised here. Director state-transitions trigger the runtime,
   8:   the runtime calls ``WSBridge.publish(...)``, and that lands on the
   9:   socket alongside our projected events. This projector therefore
  10:   returns ``None`` for events that the avatar director would map to
  11:   directives (``LLMStarted``, ``TTSStarted``, ``UserSpeechStarted``,
  12:   …) — duplication would let the same state flip twice on the wire.
  13: * ``transcript.preview`` — partial STT projection.
  14: * ``bubble.text`` — LLM token stream and final reply.
  15: * ``conversation.turn`` — replayable user/assistant history for the dashboard.
  16: * ``task.card`` — TaskSubmitted / TaskUpdatedEvent / TaskCompleted.

  18:   TTSInterrupted user-facing warnings.
