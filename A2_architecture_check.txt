  58: class BroadcastBridge:
  59:     """Process-wide WebSocket fan-out.
  60: 
  61:     Implements the structural ``WSBridge`` Protocol (``async def
  62:     publish(message: dict) -> None``) without importing it from the
  63:     avatar package â€” keeps this file free of sibling-package imports.
  64: 
  65:     Each connected socket is registered via :meth:`add_socket` /
  66:     :meth:`remove_socket` and receives every ``publish`` call. A failed
  67:     send is logged and removes the offending socket from the fan-out
  68:     set so a dead client doesn't poison the broadcast.
  69:     """
  70: 
  71:     def __init__(self) -> None:
  72:         self._sockets: set[WebSocket] = set()
  73:         self._send_locks: dict[WebSocket, asyncio.Lock] = {}
  74:         self._lock = asyncio.Lock()
  75:         self._latest_avatar: dict[str, Any] | None = None
  76:         self._latest_bubble: dict[str, Any] | None = None
  77:         self._latest_runtime: dict[str, Any] | None = None
  78:         self._latest_turn: dict[str, Any] | None = None
  79:         self._component_health: dict[str, dict[str, Any]] = {}
  80:         self._latest_tasks: dict[str, dict[str, Any]] = {}
  81:         self._conversation: dict[str, dict[str, Any]] = {}
  82: 
  83:     async def add_socket(self, ws: WebSocket) -> bool:
  84:         async with self._lock:
  85:             self._sockets.add(ws)
  86:             self._send_locks.setdefault(ws, asyncio.Lock())
  87:             replay = [message for message in (self._latest_runtime,) if message is not None]
  88:             replay.extend(self._component_health.values())
  89:             replay.extend(
  90:                 message
  91:                 for message in (self._latest_turn, self._latest_avatar, self._latest_bubble)
  92:                 if message is not None
  93:             )
  94:             replay.extend(self._conversation.values())
  95:             replay.extend(self._latest_tasks.values())
  96:         for message in replay:
  97:             if not await self.send(ws, message):
  98:                 return False
  99:         return True
 100: 
 101:     async def remove_socket(self, ws: WebSocket) -> None:
 102:         async with self._lock:
 103:             self._sockets.discard(ws)
 104:             self._send_locks.pop(ws, None)
 105: 
 106:     @property
 107:     def socket_count(self) -> int:
 108:         return len(self._sockets)
 109: 
 110:     async def publish(self, message: dict[str, Any]) -> None:
 111:         """Fan ``message`` out to every connected socket. Never raises."""
 112:         dead: list[WebSocket] = []
 113:         async with self._lock:
 114:             self._remember_unlocked(message)
 115:             sockets = list(self._sockets)
 116:         for ws in sockets:
 117:             if not await self.send(ws, message, remove_on_failure=False):
 118:                 dead.append(ws)
 119:         if dead:
 120:             async with self._lock:
 121:                 for ws in dead:
 122:                     self._sockets.discard(ws)
 123:                     self._send_locks.pop(ws, None)
 124: 
 125:     async def send(
 126:         self,
 127:         ws: WebSocket,
 128:         message: dict[str, Any],
 129:         *,
 130:         remove_on_failure: bool = True,
 131:     ) -> bool:
 132:         """Serialize all writes to one socket and absorb close/send races."""
 133: 
 134:         async with self._lock:
 135:             if ws not in self._sockets:
 136:                 return False
 137:             send_lock = self._send_locks.setdefault(ws, asyncio.Lock())
 138:         try:
 139:             async with send_lock:
 140:                 await ws.send_json(message)
 141:         except Exception as exc:
 142:             _log.info("WebSocket send dropped: type=%s error=%s", message.get("type"), exc)
 143:             if remove_on_failure:
 144:                 await self.remove_socket(ws)
 145:             return False
 146:         return True

 148:     async def remember(self, message: dict[str, Any]) -> None:
 149:         """Retain UI state so windows opened later receive a coherent view."""
 150: 
 151:         async with self._lock:
 152:             self._remember_unlocked(message)

 154:     def _remember_unlocked(self, message: dict[str, Any]) -> None:
 155:         if message.get("type") == "avatar.directive":
 156:             self._latest_avatar = dict(message)
 157:         elif message.get("type") == "runtime.state":
 158:             self._latest_runtime = dict(message)
 159:         elif message.get("type") == "turn.state":
 160:             # A rejected attempt is not the current turn and must not replace
 161:             # the authoritative state replayed to newly opened windows.
 162:             if message.get("state") != "rejected":
 163:                 self._latest_turn = dict(message)
 164:         elif message.get("type") == "component.health":
 165:             component = message.get("component")
 166:             if isinstance(component, str) and component:
 167:                 self._component_health[component] = dict(message)
 168:         elif message.get("type") == "bubble.text":
 169:             if message.get("reset") is True:
 170:                 self._latest_bubble = None
 171:             elif message.get("complete") is True:
 172:                 self._latest_bubble = dict(message)
 173:         elif message.get("type") == "conversation.turn":
 174:             turn_id = message.get("id")
 175:             if isinstance(turn_id, str) and turn_id:
 176:                 self._conversation[turn_id] = dict(message)
 177:                 while len(self._conversation) > 100:
 178:                     self._conversation.pop(next(iter(self._conversation)))
 179:         elif message.get("type") == "task.card":
 180:             update = message.get("update")
 181:             handle = update.get("handle") if isinstance(update, dict) else None
 182:             task_id = handle.get("id") if isinstance(handle, dict) else None
 183:             if isinstance(task_id, str) and task_id:
 184:                 self._latest_tasks[task_id] = dict(message)
